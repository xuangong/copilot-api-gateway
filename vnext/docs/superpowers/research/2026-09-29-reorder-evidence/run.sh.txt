#!/bin/zsh
set -euo pipefail

here=${0:A:h}
scenario=${1:-baseline}
case "$scenario" in
  baseline|upstream-pointer|upstream-keyboard|upstream-handle-keyboard|upstream-boundaries|upstream-cancel|upstream-failure|upstream-stale|mapping-draft) ;;
  *) print -u2 "usage: run.sh [baseline|upstream-pointer|upstream-keyboard|upstream-handle-keyboard|upstream-boundaries|upstream-cancel|upstream-failure|upstream-stale|mapping-draft]"; exit 2 ;;
esac
source_root=${D06_SOURCE_ROOT:-/Volumes/Projects/copilot-api-gateway/.worktrees/reference-adoption-verify/vnext}
test -f "$source_root/packages/gateway/src/app.ts"
test -s "$source_root/packages/gateway/src/shared/edge/ui-pages/dashboard-app/dist/dashboard.js.txt" || {
  print -u2 "Build the selected dashboard before browser acceptance: cd $source_root && bun run build:ui"
  exit 1
}
mkdir -p "$here/runs"
run_dir=$(mktemp -d "$here/runs/run.XXXXXX")
chmod 700 "$run_dir"
print "D06_RUN_DIR=$run_dir"
cd "$run_dir"
env -i PATH="$PATH" HOME="$HOME" D06_SOURCE_ROOT="$source_root" bun "$here/fixture.ts" > "$run_dir/server.log" 2>&1 &
fixture_pid=$!
cleanup() {
  kill "$fixture_pid" 2>/dev/null || true
  wait "$fixture_pid" 2>/dev/null || true
}
trap cleanup EXIT INT TERM
for attempt in {1..100}; do
  if [[ -s "$run_dir/connection.json" ]]; then break; fi
  if ! kill -0 "$fixture_pid" 2>/dev/null; then cat "$run_dir/server.log"; exit 1; fi
  sleep 0.1
done
test -s "$run_dir/connection.json" || { cat "$run_dir/server.log"; exit 1; }
env -i PATH="$PATH" HOME="$HOME" D06_CONNECTION="$run_dir/connection.json" node "$here/validate.mjs" "$scenario"
