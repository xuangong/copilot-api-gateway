import { pathToFileURL } from "node:url"
const root = process.env.VNEXT_PROBE_ROOT
assert(root)

const cancelled = new UpstreamExchangeCollector()
const prior = cancelled.begin({ parentCallId: "call-1", upstreamId: "upstream-1", method: "POST" })
assert(prior)
prior.observeResponse(204, [], null)
const attempt = cancelled.begin({ parentCallId: "call-2", upstreamId: "upstream-1", method: "POST" })
assert(attempt)
const source = new ReadableStream<Uint8Array>({ pull() {}, cancel() {} })
const wrapped = attempt.observeResponse(200, [], source)
assert(wrapped)
const reader = wrapped.getReader()
const pending = reader.read()
await Promise.resolve()
await reader.cancel("fixture cancellation")
await pending
const snapshot = safeUpstreamExchangesForPersistence(cancelled.finish())
assert.equal(snapshot.attempts.length, 2)
assert.equal(snapshot.attempts[0]?.response.terminal, "eof")
assert.equal(snapshot.attempts[1]?.response.terminal, "cancelled")
assert.equal(snapshot.attempts[1]?.response.totalBytes, null)
assert.equal(source.locked, false)

  override subarray(_start?: number, _end?: number): Uint8Array { throw new Error("fixture diagnostic copy failure") }
}

const direct = new Uint8Array(await new Response(makeSource()).arrayBuffer())
const isolated = new UpstreamExchangeCollector()
const observed = isolated.begin({ parentCallId: "call-3", upstreamId: "upstream-1", method: "POST" })
assert(observed)
const bytes = new Uint8Array(await new Response(observed.observeResponse(200, [], makeSource())).arrayBuffer())
assert.deepEqual(bytes, direct)
const isolatedSnapshot = safeUpstreamExchangesForPersistence(isolated.finish())

const failed = genuine.begin({ parentCallId: "call-4", upstreamId: "upstream-1", method: "POST" })

const failedSnapshot = safeUpstreamExchangesForPersistence(genuine.finish())
