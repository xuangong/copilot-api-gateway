async function* frames(events: unknown[]): AsyncGenerator<ProtocolFrame<never>> {
  for (const event of events) yield { type: "event", event: event as never }
}
async function wire(events: unknown[], source: "responses" | "messages" | "chat_completions" = "responses", target = source): Promise<string> {
  const upstream = withUpstreamTelemetry(frames(events), { protocol: source })
  const translator = target === source ? undefined : target === "responses" ? translateChatToResponsesEvents : target === "messages" ? translateResponsesEventsToMessagesEvents : translateResponsesToChatSSE
  const result = llmEventResult(upstream.events, identity, undefined, undefined, undefined, translator)
  const response = target === "responses" ? await respondResponses(result, { wantsStream: true })
    : target === "messages" ? await respondMessages(result, { wantsStream: true })
    : await respondChatCompletions(result, { wantsStream: true, includeUsageChunk: true })
  return response.text()
}

  try { for await (const event of await client(text).responses.create({ ...parameters, stream: true })) types.push(event.type) }
  catch (error) { iterationError = error instanceof Error ? error.message : String(error) }
  let finalStatus: string | undefined
  let finalError: string | undefined
  try { finalStatus = (await client(text).responses.stream(parameters).finalResponse()).status }
