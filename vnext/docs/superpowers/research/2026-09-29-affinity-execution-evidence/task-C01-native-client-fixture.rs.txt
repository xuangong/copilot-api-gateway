#[tokio::test(flavor = "multi_thread", worker_threads = 2)]
async fn vnext_c01_affinity_reasoning_roundtrip() -> anyhow::Result<()> {
    let values: serde_json::Value = serde_json::from_str(include_str!("/Volumes/Projects/copilot-api-gateway/.worktrees/reference-adoption/.superpowers/sdd/2026-09-29-reference-adoption-follow-up/task-C01-native-carriers.json"))?;
    let carrier = values["reasoning"].as_str().expect("synthetic carrier");
    let server = MockServer::start().await;
    let response_mock = mount_sse_sequence(&server, vec![
        sse(vec![ev_response_created("resp-c01-1"), json!({
            "type": "response.output_item.done", "item": {
                "id":"rs_c01", "type":"reasoning", "summary":[], "encrypted_content":carrier
            }
        }), ev_assistant_message("msg_c01", "synthetic answer"), ev_completed("resp-c01-1")]),
        sse(vec![ev_response_created("resp-c01-2"), ev_completed("resp-c01-2")]),
    ]).await;
    let test = test_codex().build(&server).await?;
    test.submit_turn("synthetic first turn").await?;
    test.submit_turn("synthetic follow-up").await?;
    let requests=response_mock.requests();
    assert_eq!(requests.len(),2);
    let input=requests[1].input();
    let reasoning=input.iter().find(|item| item["type"]=="reasoning").expect("reasoning replayed");
    assert_eq!(reasoning["encrypted_content"].as_str(),Some(carrier));
    Ok(())
}

#[tokio::test(flavor = "multi_thread", worker_threads = 2)]
async fn vnext_c01_affinity_compaction_roundtrip() -> anyhow::Result<()> {
    let values: serde_json::Value = serde_json::from_str(include_str!("/Volumes/Projects/copilot-api-gateway/.worktrees/reference-adoption/.superpowers/sdd/2026-09-29-reference-adoption-follow-up/task-C01-native-carriers.json"))?;
    let carrier=values["compaction"].as_str().expect("synthetic carrier");
    let server=MockServer::start().await;
    let response_mock=mount_sse_sequence(&server,vec![
        sse(vec![ev_response_created("resp-c01-first"),ev_completed("resp-c01-first")]),
        sse(vec![json!({"type":"response.output_item.done","item":{"type":"compaction","encrypted_content":carrier}}),ev_completed("resp-c01-compact")]),
        sse(vec![ev_response_created("resp-c01-after"),ev_completed("resp-c01-after")]),
    ]).await;
    let test=test_codex().with_auth(CodexAuth::create_dummy_chatgpt_auth_for_testing())
        .with_config(|config|{let _=config.features.enable(Feature::RemoteCompactionV2);})
        .build(&server).await?;
    test.submit_turn("synthetic before compact").await?;
    test.codex.submit(Op::Compact).await?;
    wait_for_event(&test.codex,|event|matches!(event,EventMsg::TurnComplete(_))).await;
    test.submit_turn("synthetic after compact").await?;
    let requests=response_mock.requests();
    assert_eq!(requests.len(),3);
    let input=requests[2].input();
    let compact=input.iter().find(|item|item["type"]=="compaction").expect("compaction replayed");
    assert_eq!(compact["encrypted_content"].as_str(),Some(carrier));
    Ok(())
}
