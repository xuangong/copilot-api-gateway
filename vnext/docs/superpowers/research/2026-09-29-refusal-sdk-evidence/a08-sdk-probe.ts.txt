const identity = { incomingModel: "m", model: "m", upstream: "test", modelKey: "m", cost: null }
const parts = {
  empty: [{ type: "refusal", refusal: "" }],
  mixed: [{ type: "output_text", text: "Before" }, { type: "refusal", refusal: "Denied" }, { type: "output_text", text: "After" }],
}
function result(name: "empty" | "mixed" | "failed") {
  return {
    id: `resp_${name}`, object: "response", model: "m",
    status: name === "failed" ? "failed" : "completed",
    output: name === "failed" ? [] : [{ type: "message", id: `msg_${name}`, role: "assistant", content: parts[name] }],
    error: name === "failed" ? { code: "invalid_prompt", message: "Policy denied" } : null,
    incomplete_details: null,
  }
}
async function* frames(body: ReturnType<typeof result>, expanded: boolean): AsyncGenerator<ProtocolFrame<never>> {
  if (expanded) {
    for (const frame of responsesResultToEvents(body as never)) yield frame as ProtocolFrame<never>
  } else {
    yield eventFrame({ type: body.status === "failed" ? "response.failed" : "response.completed", response: body } as never)
  }
}
async function render(target: "responses" | "chat" | "messages", name: "empty" | "mixed" | "failed", stream: boolean): Promise<Response> {
  const source = result(name)
  const data = frames(source, stream)
  if (target === "responses") return respondResponses(llmEventResult(data, identity), { wantsStream: stream })
  if (target === "chat") return respondChatCompletions(llmEventResult(
    data,
    { ...identity, translatorPair: { source: "chat_completions", hub: "responses" } },
    undefined, undefined,
    stream ? undefined : async body => translateResponsesToChatBody(body),
    stream ? translateResponsesToChatSSE : undefined,
  ), { wantsStream: stream, includeUsageChunk: false })
  return respondMessages(llmEventResult(
    data,
    { ...identity, translatorPair: { source: "messages", hub: "responses" } },
    undefined, undefined,
    stream ? undefined : async body => translateResponsesToMessagesBody(body as never),
    stream ? translateResponsesEventsToMessagesEvents : undefined,
  ), { wantsStream: stream })
}
function sdkFetch(response: Response) {
  return async () => response.clone()
}
function record(label: string, value: unknown) { console.log(JSON.stringify({ label, value })) }
for (const name of ["empty", "mixed"] as const) {

      assert.equal(events.filter(e => e.type === "content_block_start").length, name === "empty" ? 1 : 3)
