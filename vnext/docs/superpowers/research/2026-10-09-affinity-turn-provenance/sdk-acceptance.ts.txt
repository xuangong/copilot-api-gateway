type Json = Record<string, unknown>
const natural = "sdk-native-opaque-state"
const thought = "  retained native thought\n"
const answer = "plain answer"
const argumentsJson = '{"city":"Oslo"}'
function object(value: unknown): Json {
  assert(value && typeof value === "object" && !Array.isArray(value))
  return value as Json
}
function array(value: unknown): Json[] {
  assert(Array.isArray(value))
  return value.map(object)
}
function events(wire: string): Json[] {
  return wire.split("\n").filter(line => line.startsWith("data: ") && line !== "data: [DONE]").map(line => object(JSON.parse(line.slice(6))))
}
function sse(frames: Json[]): Response {
  return new Response(frames.map(frame => `event: ${frame.type}\ndata: ${JSON.stringify(frame)}\n\n`).join(""), { headers: { "content-type": "text/event-stream" } })
}
function responsesFrames(): Json[] {
  const reasoning = { type: "reasoning", id: "reasoning_sdk", summary: [{ type: "summary_text", text: thought }], encrypted_content: natural }
  const message = { type: "message", id: "message_sdk", role: "assistant", status: "completed", content: [{ type: "output_text", text: answer, annotations: [] }] }
  const call = { type: "function_call", id: "function_sdk", call_id: "call_sdk", name: "weather", arguments: argumentsJson, status: "completed" }
  const final = { id: "response_sdk", object: "response", model: "model", status: "completed", output: [reasoning, message, call], usage: { input_tokens: 3, output_tokens: 4, total_tokens: 7 } }
  return [
    { type: "response.created", response: { ...final, output: [], status: "in_progress" } },
    { type: "response.in_progress", response: { ...final, output: [], status: "in_progress" } },
    { type: "response.output_item.added", output_index: 0, item: { type: "reasoning", id: "reasoning_sdk", summary: [] } },
    { type: "response.output_item.done", output_index: 0, item: reasoning },
    { type: "response.output_item.added", output_index: 1, item: { ...message, status: "in_progress", content: [] } },
    { type: "response.content_part.added", output_index: 1, content_index: 0, item_id: "message_sdk", part: { type: "output_text", text: "", annotations: [] } },
    { type: "response.output_text.delta", output_index: 1, content_index: 0, item_id: "message_sdk", delta: "plain " },
    { type: "response.output_text.delta", output_index: 1, content_index: 0, item_id: "message_sdk", delta: "answer" },
    { type: "response.output_text.done", output_index: 1, content_index: 0, item_id: "message_sdk", text: answer },
    { type: "response.content_part.done", output_index: 1, content_index: 0, item_id: "message_sdk", part: message.content[0] },
    { type: "response.output_item.done", output_index: 1, item: message },
    { type: "response.output_item.added", output_index: 2, item: { ...call, status: "in_progress", arguments: "" } },
    { type: "response.function_call_arguments.delta", output_index: 2, item_id: "function_sdk", delta: '{"city":' },

}

function chatResponse(mode: "plain" | "tool" | "legacy"): Response {
  const base = { id: "chat_sdk", object: "chat.completion.chunk", model: "model", created: 1 }
  const chunk = (delta: Json, finish_reason: string | null = null) => ({ ...base, choices: [{ index: 0, delta, finish_reason }] })
  const frames = [chunk({ role: "assistant", content: "plain " }), chunk({ content: "answer" })]
  if (mode === "tool") frames.push(
    chunk({ tool_calls: [{ index: 0, id: "call_sdk", type: "function", function: { name: "weather", arguments: '{"city":' } }] }),
    chunk({ tool_calls: [{ index: 0, function: { arguments: '"Oslo"}' } }] }),
  )
  if (mode === "legacy") frames.push(
    chunk({ function_call: { name: "weather", arguments: '{"city":' } }),
    chunk({ function_call: { arguments: '"Oslo"}' } }),
  )
  frames.push(chunk({}, mode === "plain" ? "stop" : mode === "tool" ? "tool_calls" : "function_call"))
  return new Response(frames.map(frame => `data: ${JSON.stringify(frame)}\n\n`).join("") + "data: [DONE]\n\n", { headers: { "content-type": "text/event-stream" } })
}
async function fixture(protocol: "responses" | "messages" | "chat_completions", chatMode: "plain" | "tool" | "legacy" = "plain") {
  const { db, repo } = setupTestPlatform()
  initSocketDial(bunSocketDial)
  const pending: Promise<unknown>[] = []
  const wires: Array<{ path: string; contentType: string | null; text: string }> = []
  initBackground({ waitUntil: promise => { pending.push(promise) } })
  db.run("INSERT INTO api_keys (id, name, key, owner_id, created_at) VALUES ('key', 'sdk-test', 'sdk-fixture-only', 'owner', 'now')")
  const sent: Json[] = []
  const upstream = Bun.serve({ hostname: "127.0.0.1", port: 0, async fetch(request) {
    sent.push(object(await request.json()))
    return protocol === "chat_completions" ? chatResponse(chatMode) : sse(protocol === "responses" ? responsesFrames() : messagesFrames())
  } })
  await repo.upstreams.save({ id: "up", provider: "custom", ownerId: "owner", name: "sdk-test", enabled: true, sortOrder: 0,
    config: { name: "sdk-test", baseUrl: upstream.url.toString().replace(/\/$/, ""), authStyle: "none", endpoints: [protocol], models: ["model"] }, state: {},
    flagOverrides: {}, disabledPublicModelIds: [], proxyFallbackList: [], createdAt: "now", updatedAt: "now" })
  const gateway = Bun.serve({ hostname: "127.0.0.1", port: 0, async fetch(request) {
    const args = { raw: await request.json(), auth: { userId: "owner", apiKeyId: "key" }, obsCtx: {}, signal: request.signal }
    return protocol === "responses" ? (await serveResponses(args)).response : protocol === "messages" ? serveMessages(args) : serveChatCompletions(args)
  } })
  return {
    sent, wires,
    options: { apiKey: "synthetic-sdk-key", baseURL: gateway.url.toString(), maxRetries: 0, timeout: 10000,
      async fetch(input: string | URL | Request, init?: RequestInit) {
        const response = await fetch(input, init)
        const copy = response.clone()
        pending.push(copy.text().then(text => { wires.push({ path: String(input), contentType: copy.headers.get("content-type"), text }) }))
        return response
      },
    },
    async flush() { await Promise.all(pending) },
    async close() { gateway.stop(true); upstream.stop(true); await Promise.allSettled(pending); db.close(); __resetPlatformForTests() },
  }
}
function expectResponses(items: Json[]): void {
  assert.equal(items.length, 4)
  const [origin, reasoning, message, call] = items
  assert(origin && reasoning && message && call)
  assert.equal(origin.type, "reasoning")
  assert.deepEqual(origin.summary, [])
  assert.match(String(origin.encrypted_content), /^vnext-affinity:2:/)
  assert.match(String(reasoning.encrypted_content), /^vnext-affinity:1:/)
  assert.deepEqual(reasoning.summary, [{ type: "summary_text", text: thought }])
  assert.deepEqual(array(message.content).map(({ parsed: _sdkParsed, ...part }) => part), [{ type: "output_text", text: answer, annotations: [] }])
  assert.equal(call.arguments, argumentsJson)
}
function strictResponses(frames: Json[]): Json {
  let snapshot: Json | undefined
  const added: number[] = []
  const completed: number[] = []
  for (const [sequence, frame] of frames.entries()) {
    assert.equal(frame.sequence_number, sequence)
    if (frame.type === "response.created") {
      assert.equal(snapshot, undefined)
      snapshot = structuredClone(object(frame.response))
      assert.deepEqual(snapshot.output, [])
      continue
    }
    assert(snapshot)
    const output = snapshot.output as Json[]
    if (frame.type === "response.in_progress") {
      assert.deepEqual(object(frame.response).output, output, "Lifecycle replacement must retain exactly one already emitted origin")
      snapshot = structuredClone(object(frame.response))
    } else if (frame.type === "response.output_item.added") {
      assert.equal(frame.output_index, output.length, "Added index must extend the snapshot once")
      output.push(structuredClone(object(frame.item)))
      added.push(Number(frame.output_index))
    } else if (frame.type === "response.output_item.done") {
      const index = Number(frame.output_index)
      assert(output[index])
      assert.equal(output[index].id, object(frame.item).id)
      output[index] = structuredClone(object(frame.item))
      completed.push(index)
    } else if (frame.type === "response.completed") {
      assert.deepEqual(object(frame.response).output, output)
      snapshot = structuredClone(object(frame.response))
    } else if (typeof frame.output_index === "number") {
      const item = output[frame.output_index]
      assert(item)
      assert.equal(item.id, frame.item_id)
      if (frame.type === "response.content_part.added") {
        assert.equal(frame.content_index, array(item.content).length)
        const content = item.content as Json[]
        content.push(structuredClone(object(frame.part)))
      } else if (frame.type === "response.output_text.delta") {
        const content = (item.content as Json[])[Number(frame.content_index)]
        assert(content && content.type === "output_text")
        content.text = String(content.text) + String(frame.delta)
      } else if (frame.type === "response.output_text.done") {
        assert.equal(array(item.content)[Number(frame.content_index)]?.text, frame.text)
      } else if (frame.type === "response.content_part.done") {
        assert.deepEqual(array(item.content)[Number(frame.content_index)], frame.part)
      } else if (frame.type === "response.function_call_arguments.delta") {
        assert.equal(item.type, "function_call")
        item.arguments = String(item.arguments) + String(frame.delta)
      } else if (frame.type === "response.function_call_arguments.done") {
        assert.equal(item.arguments, frame.arguments)
      }
    }
  }
  assert(snapshot)
  assert.deepEqual(added, [0, 1, 2, 3])
  assert.deepEqual(completed, [0, 1, 2, 3])
  expectResponses(array(snapshot.output))
  return { frames: frames.length, added, completed, terminalOutputCount: array(snapshot.output).length }
}
function expectReplay(sent: Json): void {
  const serialized = JSON.stringify(sent)
  assert(!serialized.includes("vnext-affinity:"))
  assert(serialized.includes(natural))
  assert(serialized.includes(JSON.stringify(thought).slice(1, -1)))
  assert(serialized.includes(answer))
  assert(serialized.includes("Oslo"))
  assert(serialized.includes("continue"))
}
